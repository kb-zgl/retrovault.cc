# 05 — 多语言方案 (i18n)

> 框架：Nuxt.js 4 + @nuxtjs/i18n  
> 初期上线：英语（默认）+ 简体中文  
> URL策略：子目录（/zh-cn/）

---

## 语言配置

### nuxt.config.ts

```typescript
export default defineNuxtConfig({
  modules: ['@nuxtjs/i18n'],
  i18n: {
    locales: [
      { code: 'en',    iso: 'en-US', name: 'English',  file: 'en.json' },
      { code: 'zh-cn', iso: 'zh-CN', name: '简体中文', file: 'zh-cn.json' },
    ],
    defaultLocale: 'en',
    strategy: 'prefix_except_default',
    // en:    /games/slug       （无前缀）
    // zh-cn: /zh-cn/games/slug
    detectBrowserLanguage: {
      useCookie:       true,
      cookieKey:       'i18n_redirected',
      redirectOn:      'root',
      alwaysRedirect:  false,
    },
  },
})
```

## 游戏内容翻译策略

游戏标题/描述/详细介绍存储在 `game_translations` 表，按需加载。

### Nuxt 页面取值逻辑

```typescript
// composables/useGameContent.ts
export function useGameContent(game: Game) {
  const { locale } = useI18n()

  const content = computed(() => {
    // 优先用当前语言翻译
    const translation = game.translations?.[locale.value]
    if (translation?.title) return translation

    // 回退英文
    return {
      title:           game.title,
      description:     game.description,
      longDescription: game.longDescription,
      controls:        game.controls,
    }
  })

  return { content }
}
```

### 平台名本地化

```typescript
// composables/usePlatformName.ts
export function usePlatformName(platform: string) {
  const { t } = useI18n()
  const key = `platforms.${platform.toLowerCase().replace(/\s+/g, '-')}`
  return t(key, platform)  // 找不到key时回退原始值
}
```

## hreflang 输出

在 `app.vue` 或 layout 中统一处理：

```typescript
// composables/useHreflang.ts
export function useHreflang() {
  const route        = useRoute()
  const { locales }  = useI18n()

  useHead({
    link: [
      ...locales.value.map(locale => ({
        rel:      'alternate',
        hreflang: locale.iso,
        href:     `https://retrovault.online${locale.code === 'en' ? '' : `/${locale.code}`}${route.path}`,
      })),
      {
        rel:      'alternate',
        hreflang: 'x-default',
        href:     `https://retrovault.online${route.path}`,
      },
    ],
  })
}
```

## 语言切换组件

```vue
<!-- components/LanguageSwitcher.vue -->
<template>
  <div class="relative">
    <button @click="open = !open">
      {{ currentLocale.name }} ▾
    </button>
    <div v-if="open" class="absolute dropdown">
      <a
        v-for="locale in locales"
        :key="locale.code"
        :href="switchLocalePath(locale.code)"
        class="block px-4 py-2 hover:bg-gray-700"
      >
        {{ locale.name }}
      </a>
    </div>
  </div>
</template>
```
